class AWB_BorderBars {

__New() {

this._color := "Fuchsia"

this._thick := 8

this._offset := 0

this._shown := false

this._lastLayoutKey := ""



this.top := this._makeBar()

this.right := this._makeBar()

this.bottom := this._makeBar()

this.left := this._makeBar()

}



_makeBar() {

g := Gui("+AlwaysOnTop -Caption +ToolWindow +E0x20 +E0x80000", "AWB_BorderBar")

g.BackColor := this._color



; Force-create a real HWND now (hidden), so g.Hwnd is valid.

try {

g.Show("Hide")

} catch as e {

; ignore creation errors

}



; Layering after creation

try {

DllCall("SetLayeredWindowAttributes", "ptr", g.Hwnd, "ptr", 0, "uchar", 255, "uint", 0x02)

} catch as e {

; ignore

}

return g

}



_isAlive(g) {

hwnd := 0

try {

hwnd := g.Hwnd

} catch as e {

hwnd := 0

}

return hwnd && !!DllCall("IsWindow", "ptr", hwnd, "int")

}



_ensureAlive() {

static names := ["top","right","bottom","left"]

for name in names {

bar := this.%name%

if (!this._isAlive(bar)) {

; Recreate and preserve style

nb := this._makeBar()

try {

nb.BackColor := this._color

} catch as e {

; ignore

}

this.%name% := nb

this._lastLayoutKey := ""

}

}

}



SetStyle(color, thickness, offset := 0) {

if (this._color = color && this._thick = thickness && this._offset = offset)

return



this._color := color

this._thick := thickness

this._offset := offset



this._ensureAlive()

for bar in [this.top, this.right, this.bottom, this.left] {

try {

bar.BackColor := color

} catch as e {

; ignore

}

}

this._lastLayoutKey := ""

}



SetOwner(ownerHwnd) {

global AWB_BarsOwner

if (AWB_BarsOwner = ownerHwnd)

return

if (!ownerHwnd || !DllCall("IsWindow", "ptr", ownerHwnd, "int"))

return



this._ensureAlive()

for bar in [this.top, this.right, this.bottom, this.left] {

hwnd := 0

try {

hwnd := bar.Hwnd

} catch as e {

hwnd := 0

}

if (!hwnd)

continue



; GWLP_HWNDPARENT = -8

if (A_PtrSize = 8) {

try {

DllCall("SetWindowLongPtr", "ptr", hwnd, "int", -8, "ptr", ownerHwnd)

} catch as e {

; ignore

}

} else {

try {

DllCall("SetWindowLong", "ptr", hwnd, "int", -8, "int", ownerHwnd)

} catch as e {

; ignore

}

}

}

AWB_BarsOwner := ownerHwnd

}



UpdateRect(rect) {

if !rect || rect.w <= 0 || rect.h <= 0

return



this._ensureAlive()



t := this._thick

o := this._offset

x := rect.x + o

y := rect.y + o

w := Max(0, rect.w - 2*o)

h := Max(0, rect.h - 2*o)

if (w <= 0 || h <= 0)

return



global AWB_ForceTopmostZ

HWND_TOPMOST := -1

HWND_NOTOPMOST := -2

zBand := AWB_ForceTopmostZ ? HWND_TOPMOST : HWND_NOTOPMOST

SWP := 0x0010 | 0x0040 | 0x4000 ; NOACTIVATE | SHOWWINDOW | ASYNCPOS



layoutKey := x "|" y "|" w "|" h "|" t "|" zBand

if (this._shown && this._lastLayoutKey = layoutKey)

return



; Safe SetWindowPos calls for each bar

try {

DllCall("SetWindowPos", "ptr", this.top.Hwnd, "ptr", zBand, "int", x, "int", y, "int", w, "int", t, "uint", SWP)

} catch as e {

}

try {

DllCall("SetWindowPos", "ptr", this.right.Hwnd, "ptr", zBand, "int", x+w-t, "int", y, "int", t, "int", h, "uint", SWP)

} catch as e {

}

try {

DllCall("SetWindowPos", "ptr", this.bottom.Hwnd, "ptr", zBand, "int", x, "int", y+h-t, "int", w, "int", t, "uint", SWP)

} catch as e {

}

try {

DllCall("SetWindowPos", "ptr", this.left.Hwnd, "ptr", zBand, "int", x, "int", y, "int", t, "int", h, "uint", SWP)

} catch as e {

}

this._lastLayoutKey := layoutKey

}



ShowIfHidden() {

if this._shown

return

this._ensureAlive()

for bar in [this.top, this.right, this.bottom, this.left] {

try {

bar.Show("NA")

} catch as e {

; ignore

}

}

this._shown := true

this._lastLayoutKey := ""

}



Hide() {

this._ensureAlive()

for bar in [this.top, this.right, this.bottom, this.left] {

try {

bar.Hide()

} catch as e {

; ignore "no window"

}

}

this._shown := false

this._lastLayoutKey := ""

}

}

return -1

}

return 0

if (val = 2)

return 1

return 2

}

return -1

}

AWB_Dragon_IsRunning() {

; Prefer PID from live Dragon UI windows to avoid stale/suspended process IDs.

bestPid := 0

try {

for _, hwnd in WinGetList("ahk_exe natspeak.exe") {

if !hwnd

continue

pid := 0

try {

pid := WinGetPID("ahk_id " hwnd)

} catch {

pid := 0

}

if (pid > bestPid)

bestPid := pid

}

} catch {

}

if bestPid

return bestPid



; Fallback to known DragonBar hosts (some builds expose this separately).

for crit in ["ahk_class DragonBar", "DragonBar"] {

hwnd := 0

try {

hwnd := WinExist(crit)

} catch {

hwnd := 0

}

if !hwnd

continue

pid := 0

try {

pid := WinGetPID("ahk_id " hwnd)

} catch {

pid := 0

}

if pid

return pid

}



; No live Dragon UI windows found.

return 0

}

}

; Gather likely Dragon windows (named candidates + all natspeak top-level windows).

candidates := ["ahk_class DragonBar", "DragonBar", "ahk_exe natspeak.exe"]

seen := Map()

} catch {

hwnd := 0

}

seen[hwnd] := true

hwnds.Push(hwnd)

}

}



try {

for _, h in WinGetList("ahk_exe natspeak.exe") {

if (h && !seen.Has(h)) {

seen[h] := true

hwnds.Push(h)

}

}

} catch {

}



for _, hwnd in hwnds {

t := ""

try {

} catch {

t := ""

}

if (t != "")

texts.Push(t)



cls := ""

try {

cls := WinGetClass("ahk_id " hwnd)

} catch {

cls := ""

}

if (cls != "")

texts.Push(cls)

}



; Inspect collected UI text for mic-state tokens.

for _, t in texts {

tu := StrUpper(t)

if InStr(tu, "ASLEEP") || InStr(tu, "SLEEPING")

return 2

if InStr(tu, "NOT LISTENING") || InStr(tu, "MICROPHONE OFF")

return 0

if InStr(tu, "LISTENING") || InStr(tu, "MICROPHONE ON")

return 1

if RegExMatch(tu, "i)\bMIC(ROPHONE)?\s+OFF\b")

return 0

if RegExMatch(tu, "i)\bMIC(ROPHONE)?\s+ON\b")

return 1

}

return -1

}

AWB_WinEventCallback(hook, event, hwnd, idObject := 0, idChild := 0, dwEventThread := 0, dwmsEventTime := 0) {

global AWB_LastHwnd, AWB_LastRect, AWB_LastClass

global AWB_SizingHwnd, AWB_SizingTimer, AWB_TrackTimer

global AWB_EpochCounter, AWB_TargetEpoch, AWB_CurrentTarget

global AWB_OfficeDrag, AWB_SizingStyle, AWB_ForceTopmostZ



Critical -1

try {

if (idObject != AWB_OBJID_WINDOW || idChild != AWB_CHILDID_SELF) {

return

}



; Foreground landing

if (event = AWB_EVENT_SYSTEM_FOREGROUND) {

if AWB_IsWindow(hwnd) {

prevTarget := AWB_CurrentTarget

tentative := AWB_GetHighlightTarget(hwnd)

if !AWB_IsWindow(tentative) {

tentative := hwnd

}

if (AWB_IsWindow(prevTarget) && prevTarget != tentative) {

; Invalidate pending refresh timers for the previous target.

AWB_TargetEpoch[prevTarget] := ++AWB_EpochCounter

}

AWB_CurrentTarget := tentative

epoch := ++AWB_EpochCounter

AWB_TargetEpoch[tentative] := epoch



; Teams: immediate best-effort draw + retries

if (AWB_IsTeamsWindow(tentative) && AWB_ShouldRenderWindow(tentative)) {

rect := AWB_GetStableRect(tentative)

if rect {

style := AWB_PickStyle(tentative)

if (!style || style = 0)

style := { Color: "Fuchsia", Thickness: 8, Offset: 0 }

AWB_EnsureBars()

AWB_Bars.SetOwner(tentative)

AWB_Bars.SetStyle(style.Color, style.Thickness, style.HasProp("Offset") ? style.Offset : 0)

AWB_Bars.ShowIfHidden()

AWB_Bars.UpdateRect(rect)

AWB_MarkDrawn()

AWB_LastHwnd := tentative

AWB_LastRect := { x: rect.x, y: rect.y, w: rect.w, h: rect.h }

}

AWB_ScheduleRefresh(tentative, [20, 60, 120, 240, 480, 960], epoch)

} else {

AWB_ScheduleRefresh(tentative, [20, 60, 120, 240, 480, 960], epoch)

}

}

}



; CREATE / DESTROY

if (event = AWB_EVENT_OBJECT_CREATE) {

if AWB_IsWindow(hwnd) {

if (AWB_CurrentTarget && !AWB_IsRelatedToCurrentTarget(hwnd)) {

return

}

epoch := ++AWB_EpochCounter

AWB_TargetEpoch[hwnd] := epoch

AWB_ScheduleRefresh(hwnd, [30, 80, 160, 320, 640], epoch)

}

return ; prevent fall-through immediate render

}



if (event = AWB_EVENT_OBJECT_DESTROY) {

if (AWB_LastHwnd = hwnd) {

AWB_HideBars()

}

if (AWB_CurrentTarget = hwnd) {

AWB_CurrentTarget := 0

}

if (AWB_TargetEpoch.Has(hwnd)) {

AWB_TargetEpoch.Delete(hwnd)

}

return

}



; LOCATIONCHANGE throttling

if (event = AWB_EVENT_OBJECT_LOCATIONCHANGE) {

; If Office drag is in progress, ignore unrelated location changes

if (AWB_OfficeDrag && hwnd != AWB_SizingHwnd) {

return

}

if (AWB_SizingHwnd = 0 && !AWB_IsRelatedToCurrentTarget(hwnd)) {

return

}

}



; SHOW/HIDE

if (event = AWB_EVENT_OBJECT_SHOW || event = AWB_EVENT_OBJECT_HIDE) {

if (AWB_CurrentTarget && AWB_IsWindow(AWB_CurrentTarget)) {

epoch := ++AWB_EpochCounter

AWB_TargetEpoch[AWB_CurrentTarget] := epoch

AWB_ScheduleRefresh(AWB_CurrentTarget, [120, 300], epoch)

}

return ; prevent fall-through immediate render

}



; Minimize/restore

if (event = AWB_EVENT_SYSTEM_MINIMIZESTART) {

AWB_HideBars()

return

}

if (event = AWB_EVENT_SYSTEM_MINIMIZEEND) {

tgt := AWB_GetRootOwner(hwnd)

if !AWB_IsWindow(tgt) {

tgt := hwnd

}

epoch := ++AWB_EpochCounter

AWB_TargetEpoch[tgt] := epoch

AWB_CurrentTarget := tgt

AWB_ScheduleRefresh(tgt, [40, 120, 240, 480], epoch)

return

}



; ---- Live move/size handling ----

if (event = AWB_EVENT_SYSTEM_MOVESIZESTART) {

if !AWB_IsWindow(hwnd) {

return

}

; Use GA_ROOT for Office (Excel) to get the real frame; GA_ROOTOWNER otherwise

if (AWB_IsOfficeWindow(hwnd)) {

root := DllCall("GetAncestor", "ptr", hwnd, "uint", AWB_GA_ROOT, "ptr")

if !AWB_IsWindow(root)

root := hwnd

} else {

root := AWB_GetRootOwner(hwnd)

if !AWB_IsWindow(root)

root := hwnd

}

if !AWB_ShouldRenderWindow(root) {

AWB_HideBars()

return

}

AWB_SizingHwnd := root



; Office drag mode ON

if (AWB_IsOfficeWindow(root)) {

AWB_OfficeDrag := true

AWB_ForceTopmostZ := true

} else {

AWB_OfficeDrag := false

AWB_ForceTopmostZ := false

}



rect := AWB_IsOfficeWindow(root) ? AWB_GetOfficeFrameRect(root) : AWB_GetWindowRectUser(root)

if rect {

style := AWB_PickStyle(root)

if (!style || style = 0)

style := { Color: "Fuchsia", Thickness: 8, Offset: 0 }

AWB_SizingStyle := style ; lock style during drag



AWB_EnsureBars()

AWB_Bars.SetOwner(root)

AWB_Bars.SetStyle(style.Color, style.Thickness, style.HasProp("Offset") ? style.Offset : 0)

AWB_Bars.ShowIfHidden()

AWB_Bars.UpdateRect(rect)

AWB_MarkDrawn()



AWB_LastHwnd := root

AWB_LastRect := { x: rect.x, y: rect.y, w: rect.w, h: rect.h }

AWB_LastClass := AWB_SafeWinGetClass(root)



AWB_CurrentTarget := root

AWB_EpochCounter += 1

AWB_TargetEpoch[root] := AWB_EpochCounter



if (AWB_SizingTimer = 0) {

SetTimer(AWB_SizingTick, 15)

AWB_SizingTimer := 1

}

if (AWB_TrackTimer = 0) {

SetTimer(AWB_TrackTick, 33)

AWB_TrackTimer := 1

}

}

return

}



if (event = AWB_EVENT_SYSTEM_MOVESIZEEND) {

endHwnd := hwnd

if (AWB_IsOfficeWindow(endHwnd)) {

root := DllCall("GetAncestor", "ptr", endHwnd, "uint", AWB_GA_ROOT, "ptr")

if !AWB_IsWindow(root)

root := endHwnd

} else {

root := AWB_GetRootOwner(endHwnd)

if !AWB_IsWindow(root)

root := endHwnd

}

if (AWB_SizingHwnd = root) {

AWB_SizingHwnd := 0

}

if (AWB_SizingTimer) {

SetTimer(AWB_SizingTick, 0)

AWB_SizingTimer := 0

}

; Office drag mode OFF

AWB_OfficeDrag := false

AWB_ForceTopmostZ := false

AWB_SizingStyle := 0



if !AWB_IsWindow(root) {

AWB_HideBars()

return

}



AWB_EpochCounter += 1

AWB_TargetEpoch[root] := AWB_EpochCounter

AWB_CurrentTarget := root

AWB_ScheduleRefresh(root, [60, 160, 320, 640], AWB_TargetEpoch[root])

; fall-through to normal processing

}



; ---- STRICT RENDER GATE: only render for current target / key events ----

if !AWB_ShouldRenderNow(hwnd, event) {

return

}



target := AWB_GetHighlightTarget(hwnd)

if !target {

AWB_HideBarsSoft()

return

}



if !AWB_IsWindowEligible(target) {

target := AWB_GetRootOwner(target)

}

if !AWB_IsWindow(target) {

AWB_HideBarsSoft()

return

}



mm := AWB_SafeWinGetMinMax(target) ; -1=min, 0=normal, 1=max

if !AWB_ShouldRenderWindow(target, 0, mm) {

AWB_HideBarsSoft()

return

}



; Office first-entry deferral

if (AWB_IsOfficeWindow(target) && mm = 0 && AWB_IsOfficeUnstableFirstEntry(target)) {

AWB_EpochCounter += 1

AWB_TargetEpoch[target] := AWB_EpochCounter

AWB_CurrentTarget := target

AWB_ScheduleRefresh(target, [80, 200, 400, 800], AWB_TargetEpoch[target])

return

}



rect := AWB_GetStableRect(target)

if !rect {

; During Office drags: NEVER hide while dragging the sizing target

if (AWB_OfficeDrag && AWB_SizingHwnd = target) {

return

}

if (AWB_LastHwnd != target) {

AWB_HideBarsSoft()

}

epoch := ++AWB_EpochCounter

AWB_TargetEpoch[target] := epoch

AWB_CurrentTarget := target

AWB_ScheduleRefresh(target, [120, 300, 600], epoch)

return

}

if !AWB_ShouldRenderWindow(target, rect, mm) {

AWB_HideBarsSoft()

return

}



; Use the locked style during an Office drag (avoid fallback "A" swaps)

style := (AWB_OfficeDrag && AWB_SizingHwnd = target && AWB_SizingStyle)

? AWB_SizingStyle

: AWB_PickStyle(target)

if (!style || style = 0)

style := { Color: "Fuchsia", Thickness: 8, Offset: 0 }



AWB_EnsureBars()

AWB_Bars.SetOwner(target)

; Only set style if not in Office drag OR style changed outside a drag

if !(AWB_OfficeDrag && AWB_SizingHwnd = target) {

AWB_Bars.SetStyle(style.Color, style.Thickness, style.HasProp("Offset") ? style.Offset : 0)

}

AWB_Bars.ShowIfHidden()



if (AWB_LastHwnd != target

|| AWB_LastRect.x != rect.x || AWB_LastRect.y != rect.y

|| AWB_LastRect.w != rect.w || AWB_LastRect.h != rect.h) {



AWB_Bars.UpdateRect(rect)

AWB_LastHwnd := target

AWB_LastRect := { x: rect.x, y: rect.y, w: rect.w, h: rect.h }

}

AWB_MarkDrawn()



AWB_CurrentTarget := target

AWB_EpochCounter += 1

AWB_TargetEpoch[target] := AWB_EpochCounter



if (AWB_IsOfficeWindow(target) || (AWB_LastHwnd && AWB_IsOfficeWindow(AWB_LastHwnd))) {

epoch := AWB_TargetEpoch[target]

AWB_ScheduleRefresh(target, [80, 200, 400, 800], epoch)

}



if (AWB_TrackTimer = 0) {

SetTimer(AWB_TrackTick, 33)

AWB_TrackTimer := 1

}



} catch as e {

; Uncomment for diagnostics:

; MsgBox "Border error: " e.Message

}

}

AWB_ParseRulesText(text, &appRules, &dragonRules, &errorText := "") {

app := []

dragon := []

errors := []

section := ""



Loop Parse, text, "`n", "`r" {

lineNo := A_Index

raw := A_LoopField

line := Trim(raw)

if (line = "")

continue



first := SubStr(line, 1, 1)

if (first = "#" || first = ";")

continue



if RegExMatch(line, "^\[(.+)\]$", &mSec) {

sec := StrUpper(Trim(mSec[1]))

if (sec = "AWB_RULES") {

section := "app"

} else if (sec = "AWB_DRAGON_RULES") {

section := "dragon"

} else {

errors.Push("Line " lineNo ": Unknown section [" mSec[1] "].")

section := ""

}

continue

}



if (section = "") {

errors.Push("Line " lineNo ": Rule is outside a section.")

continue

}



cols := InStr(raw, "|") ? StrSplit(raw, "|") : StrSplit(raw, A_Tab)

for i, col in cols

cols[i] := Trim(col)

if (cols.Length < 3 || cols.Length > 4) {

errors.Push("Line " lineNo ": Expected 3 or 4 columns separated by | or TAB.")

continue

}



selector := Trim(cols[1])

color := Trim(cols[2])

thickTxt := Trim(cols[3])

offsetTxt := (cols.Length >= 4) ? Trim(cols[4]) : "0"



if (selector = "") {

errors.Push("Line " lineNo ": Selector is empty.")

continue

AWB_ClampRectToBounds(rect, bounds, tolPx) {

if !rect || !bounds

return rect



l := rect.x

t := rect.y

r := rect.x + rect.w

b := rect.y + rect.h



bl := bounds.x

bt := bounds.y

br := bounds.x + bounds.w

bb := bounds.y + bounds.h



if (Abs(l - bl) <= tolPx)

l := bl

if (Abs(t - bt) <= tolPx)

t := bt

if (Abs(r - br) <= tolPx)

r := br

if (Abs(b - bb) <= tolPx)

b := bb



w := r - l

h := b - t

if (w <= 0 || h <= 0)

return rect

return { x: l, y: t, w: w, h: h }

}



AWB_NormalizeOfficeFrameRect(hwnd, rect, mm := "") {

if !rect

return rect

if (mm = "")

mm := AWB_SafeWinGetMinMax(hwnd)



dpi := AWB_GetDpiForWindowSafe(hwnd)

tolPx := Max(4, Ceil(12 * dpi / 96))



; Maximized windows should align to monitor work area, not outer resize margins.

if (mm = 1) {

work := AWB_GetMonitorWorkRect(hwnd)

if work

return AWB_ClampRectToBounds(rect, work, tolPx)

return rect

}



; Near-fullscreen Office windows are often off by a few px on one or more edges.

mon := AWB_GetMonitorRect(hwnd)

if !mon

return rect

if (rect.w >= Floor(mon.w * 0.90) && rect.h >= Floor(mon.h * 0.90)) {

return AWB_ClampRectToBounds(rect, mon, tolPx)

}

return rect

}



AWB_IsFullscreenWindow(hwnd, rect := 0, mm := "") {

if !AWB_IsWindow(hwnd) {

return false

}

if (mm = "")

mm := AWB_SafeWinGetMinMax(hwnd)

if (mm = -1 || mm = 1) {

return false

}



if !rect {

rect := AWB_GetWindowRectDWM(hwnd)

if !rect

rect := AWB_GetWindowRectUser(hwnd)

}

if (!rect || rect.w <= 0 || rect.h <= 0) {

return false

}



mon := AWB_GetMonitorRect(hwnd)

if !mon {

return false

}



dpi := AWB_GetDpiForWindowSafe(hwnd)

tol := Max(2, Ceil(2 * dpi / 96))

return (Abs(rect.x - mon.x) <= tol

&& Abs(rect.y - mon.y) <= tol

&& Abs((rect.x + rect.w) - (mon.x + mon.w)) <= tol

&& Abs((rect.y + rect.h) - (mon.y + mon.h)) <= tol)

}

AWB_GetWindowRectUser(hwnd) {

if !AWB_IsWindow(hwnd) {

return 0

}

try {

if WinGetPos(&x, &y, &w, &h, "ahk_id " hwnd) {

if (w > 0 && h > 0)

return { x: x, y: y, w: w, h: h }

}

} catch {

; fall through to GDI/DWM

}

r := AWB_GetWindowRectGDI(hwnd)

if r {

return r

}

r := AWB_GetWindowRectDWM(hwnd)

}



AWB_GetOfficeFrameRect(hwnd) {

mm := AWB_SafeWinGetMinMax(hwnd)

rUSR := AWB_GetWindowRectUser(hwnd)

rDWM := AWB_GetWindowRectDWM(hwnd)



if (rDWM && rUSR) {

dw := rUSR.w - rDWM.w

dh := rUSR.h - rDWM.h



; If DWM reports a larger frame than User32, prefer User32 for safety.

if (dw < 0 || dh < 0) {

return AWB_NormalizeOfficeFrameRect(hwnd, rUSR, mm)

}



; Prefer DWM bounds for static Office windows: it follows the visible frame

; and avoids invisible resize borders.

if (dw >= 0 && dh >= 0 && dw <= 64 && dh <= 64) {

return AWB_NormalizeOfficeFrameRect(hwnd, rDWM, mm)

}



; If DWM is clearly in a transient state, trust User32 for this frame.

if (Abs(dw) > 96 || Abs(dh) > 96) {

return AWB_NormalizeOfficeFrameRect(hwnd, rUSR, mm)

}



return AWB_NormalizeOfficeFrameRect(hwnd, rDWM, mm)

}

best := rDWM ? rDWM : rUSR

return best ? AWB_NormalizeOfficeFrameRect(hwnd, best, mm) : 0

}



; DPI-aware first-entry instability test for Office shells

AWB_IsOfficeUnstableFirstEntry(hwnd) {

rUSR := AWB_GetWindowRectUser(hwnd)

if !rUSR {

return false

}

rDWM := AWB_GetWindowRectDWM(hwnd)

if !rDWM {

return false

}

dx := Abs(rDWM.w - rUSR.w)

dy := Abs(rDWM.h - rUSR.h)

ox := Abs(rDWM.x - rUSR.x)

oy := Abs(rDWM.y - rUSR.y)

diffPx := 24

dpi := AWB_GetDpiForWindowSafe(hwnd)

if (dpi)

diffPx := Ceil(diffPx * dpi / 96)

return (dx > diffPx || dy > diffPx || ox > Ceil(diffPx / 2) || oy > Ceil(diffPx / 2))

}



AWB_GetStableRect(hwnd) {

global AWB_SizingHwnd

if !AWB_IsWindow(hwnd) {

return 0

}



if (AWB_SizingHwnd = hwnd) {

r := AWB_GetWindowRectUser(hwnd)

return r ? r : 0

}



; OFFICE: prefer normalized visible frame rect

if AWB_IsOfficeWindow(hwnd) {

rOffice := AWB_GetOfficeFrameRect(hwnd)

return rOffice ? rOffice : 0

}



; Non-Office: prefer DWM, clamp to User32 if large divergence

rDWM := AWB_GetWindowRectDWM(hwnd)

rUSR := AWB_GetWindowRectUser(hwnd)



if (!rDWM && !rUSR) {

return 0

}

if (rDWM && rUSR) {

dx := Abs(rDWM.w - rUSR.w)

dy := Abs(rDWM.h - rUSR.h)

return (dx > 16 || dy > 16) ? rUSR : rDWM

}

return rDWM ? rDWM : rUSR

}

; =====================================================================

; One-shot Refresh (debounce) - epoch-guarded

; =====================================================================

AWB_ScheduleRefresh(hwnd, ms, epoch := 0) {

if !AWB_IsWindow(hwnd)

return

if (epoch = 0) {

if (AWB_TargetEpoch.Has(hwnd))

epoch := AWB_TargetEpoch[hwnd]

else

epoch := 1

}

if IsObject(ms) {

for _, delay in ms

SetTimer(Func("AWB_RefreshProc").Bind(hwnd, epoch), -delay)

} else {

SetTimer(Func("AWB_RefreshProc").Bind(hwnd, epoch), -ms)

}

}



AWB_RefreshProc(hwnd, epoch) {

global AWB_LastHwnd, AWB_LastRect, AWB_Bars, AWB_TargetEpoch, AWB_CurrentTarget



if (!AWB_TargetEpoch.Has(hwnd) || AWB_TargetEpoch[hwnd] != epoch)

return



if !AWB_IsWindow(hwnd)

return



activeTarget := AWB_GetActiveTarget()

if (!activeTarget || hwnd != activeTarget)

return

AWB_CurrentTarget := activeTarget



if !AWB_ShouldShowForWindow(hwnd) {

if (AWB_LastHwnd = hwnd || AWB_CurrentTarget = hwnd)

AWB_HideBarsSoft()

return

}



rect := AWB_GetStableRect(hwnd)

if !rect {

if (AWB_LastHwnd = hwnd || AWB_CurrentTarget = hwnd)

AWB_HideBarsSoft()

return

}

if !AWB_ShouldShowForWindow(hwnd, rect) {

if (AWB_LastHwnd = hwnd || AWB_CurrentTarget = hwnd)

AWB_HideBarsSoft()

return

}



style := AWB_PickStyle(hwnd)

if (!style || style = 0)

style := { Color: "Fuchsia", Thickness: 8, Offset: 0 }



AWB_EnsureBars()

AWB_Bars.SetOwner(hwnd)

AWB_Bars.SetStyle(style.Color, style.Thickness, style.HasProp("Offset") ? style.Offset : 0)

AWB_Bars.ShowIfHidden()

AWB_Bars.UpdateRect(rect)

AWB_MarkDrawn()



AWB_LastHwnd := hwnd

AWB_LastRect := { x: rect.x, y: rect.y, w: rect.w, h: rect.h }

AWB_CurrentTarget := hwnd

}

}



AWB_TrackTick(*) {

return

AWB_GetToolbarButtonText(tb, idCmd) {

TB_GETBUTTONTEXTW := 0x044B



; First, try TB_GETBUTTONTEXTW to query length.

len := DllCall("SendMessage"

, "ptr", tb

, "uint", TB_GETBUTTONTEXTW

, "int", idCmd ; wParam

, "ptr", 0 ; lParam = NULL to get length

, "int") ; return int (length or -1)

if (len != -1) {

buf := Buffer(2 * (len + 2), 0) ; UTF-16 buffer

got := DllCall("SendMessage"

, "ptr", tb

, "uint", TB_GETBUTTONTEXTW

, "int", idCmd

, "ptr", buf.Ptr

, "int")

if (got != -1) {

txt := StrGet(buf, "UTF-16")

if (txt != "")

return txt

}

}



; Fallback: TB_GETBUTTONINFOW (fills text into pszText)

TB_GETBUTTONINFOW := 0x043F

TBIF_TEXT := 0x00000002



TBBI_size := (A_PtrSize = 8) ? 48 : 40

bi := Buffer(TBBI_size, 0)

NumPut("UInt", TBBI_size, bi, 0) ; cbSize

NumPut("UInt", TBIF_TEXT, bi, 4) ; dwMask

NumPut("Int", idCmd, bi, 8) ; idCommand



; Compute portable offsets (x86/x64)

off_dwData := (A_PtrSize = 8) ? 24 : 20

off_pszText := off_dwData + A_PtrSize

off_cchText := off_pszText + A_PtrSize



bufChars := 260

txtBuf := Buffer(2 * bufChars, 0)

NumPut("Ptr", txtBuf.Ptr, bi, off_pszText)

NumPut("Int", bufChars, bi, off_cchText)



got := DllCall("SendMessage"

, "ptr", tb

, "uint", TB_GETBUTTONINFOW

, "int", idCmd ; wParam

, "ptr", bi.Ptr ; lParam = pointer to TBBUTTONINFOW

, "int") ; return int (index or -1)

if (got = -1)

return ""



return StrGet(txtBuf, "UTF-16")

}



AWB_ScanToolbarForDragon(tb) {

if !tb

return { found: false }



TB_BUTTONCOUNT := 0x0418

TB_GETBUTTON := 0x0417



count := DllCall("SendMessage"

, "ptr", tb

, "uint", TB_BUTTONCOUNT

, "ptr", 0

, "ptr", 0

, "int")

if (count <= 0)

return { found: false }



TBBUTTON_size := (A_PtrSize = 8) ? 32 : 20

btn := Buffer(TBBUTTON_size, 0)

idCmd_off := (A_PtrSize = 8) ? 8 : 4



Loop count {

idx := A_Index - 1



ok := DllCall("SendMessage"

, "ptr", tb

, "uint", TB_GETBUTTON

, "int", idx ; wParam = index

, "ptr", btn.Ptr ; lParam = pointer to TBBUTTON buffer

, "int")

if !ok

continue



idCmd := NumGet(btn, idCmd_off, "Int")

text := AWB_GetToolbarButtonText(tb, idCmd)

if (text = "")

continue



tu := StrUpper(text)

if InStr(tu, "DRAGON") || InStr(tu, "NUANCE") || InStr(tu, "DPI") {

mic := -1

if InStr(tu, "ASLEEP")

mic := 2

else if InStr(tu, "OFF") || InStr(tu, "NOT LISTENING")

mic := 0

else if InStr(tu, "ON") || InStr(tu, "LISTENING") || InStr(tu, "ACTIVE")

mic := 1

return { found: true, text: text, mic: mic }

}

}

return { found: false }

}
